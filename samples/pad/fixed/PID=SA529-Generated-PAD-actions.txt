
FUNCTION GetAttachmentNames GLOBAL
    SET txt_AttachmentNames TO $''''''
    LOOP FOREACH CurrentItem IN In_dtb_Attachments
        IF In_flg_SetFlag THEN
            Text.AppendLine Text: txt_AttachmentNames LineToAppend: $''';\\n%CurrentItem['Attachment']%''' Result=> txt_AttachmentNames
        ELSE
            Text.AppendLine Text: txt_AttachmentNames LineToAppend: $''';\\n%CurrentItem['Name']%''' Result=> txt_AttachmentNames
        END
    END
    IF NOT IsEmpty(txt_AttachmentNames) THEN
        Text.GetSubtext.GetSubtext CharacterPosition: 1 NumberOfChars: txt_AttachmentNames.Length - 1 Text: txt_AttachmentNames Subtext=> txt_AttachmentNames
        Text.Trim Text: txt_AttachmentNames TrimOption: Text.TrimOption.Both TrimmedText=> Out_txt_AttachmentNames
    END
END FUNCTION

FUNCTION Write_To_Final_Reports GLOBAL
    Variables.FilterDataTable DataTable: In_dtb_InputFiles FilterParameters: { ^['RULE', 'COLUMN', 'OPERATOR', 'VALUE'], [$'''And''', $'''Return Code''', $'''StartsWith''', In_obj_ConfigFileData['Success Return Code']] } FilteredDataTable=> dtb_SuccessFiles
    Variables.FilterDataTable DataTable: In_dtb_InputFiles FilterParameters: { ^['RULE', 'COLUMN', 'OPERATOR', 'VALUE'], [$'''And''', $'''Return Code''', $'''NotStartsWith''', In_obj_ConfigFileData['Success Return Code']] } FilteredDataTable=> dtb_ExceptionFiles
    SET num_SuccessCount TO dtb_SuccessFiles.RowsCount
    SET num_ExceptionCount TO dtb_ExceptionFiles.RowsCount
    IF num_ExceptionCount = 0 THEN
        SET Out_txt_FullyProcessed TO $'''True'''
        CALL Get_Processed_Attachment_Names In_dtb_Attachments: dtb_SuccessFiles In_flg_SetFlag: False Out_txt_AttachmentNames=> txt_ProcessedAttachments
        CALL Write_To_Success_Report In_txt_ProcessedReports: txt_ProcessedAttachments
        EXIT
    END
    IF num_ExceptionCount > 0 THEN
        CALL Get_Unprocessed_Attachment_Names In_dtb_Attachments: dtb_ExceptionFiles In_flg_SetFlag: False Out_txt_AttachmentNames=> txt_UnprocessedAttachments
        CALL Get_Processed_Attachment_Names In_dtb_Attachments: dtb_SuccessFiles In_flg_SetFlag: False Out_txt_AttachmentNames=> txt_ProcessedAttachments
        IF IsEmpty(Trim(txt_ProcessedAttachments)) THEN
            SET txt_ProcessedAttachments TO $'''None'''
        END
    END
    SET txt_ProcessedAttachments TO $''''''
    SET txt_UnprocessedAttachments TO $''''''
    SET txt_ExceptionReason TO $''''''
    LOOP FOREACH CurrentExceptionRow IN dtb_ExceptionFiles
        IF NOT IsEmpty(Trim(CurrentExceptionRow['Return Code'])) THEN
            Text.AppendLine Text: txt_ExceptionReason LineToAppend: $''';\\n%CurrentExceptionRow['Name']%  -  %CurrentExceptionRow['Return Code']%''' Result=> txt_ExceptionReason
        ELSE
            Text.AppendLine Text: txt_ExceptionReason LineToAppend: $''';\\n%CurrentExceptionRow['Name']%  -  %In_txt_ExceptionDetail%''' Result=> txt_ExceptionReason
        END
    END
    Text.Trim Text: txt_ExceptionReason TrimOption: Text.TrimOption.Both TrimmedText=> txt_ExceptionReason
    Text.GetSubtext.GetSubtextFrom CharacterPosition: 1 Text: txt_ExceptionReason Subtext=> txt_ExceptionReason
    CALL Write_To_Exception_Report In_txt_ExceptionAttachments: txt_UnprocessedAttachments In_txt_ProcessedAttachments: txt_ProcessedAttachments In_txt_ExceptionReason: txt_ExceptionReason In_txt_ExceptionReasonColumn: $'''J'''
    SET Out_txt_FullyProcessed TO $'''False'''
    # Add your processing logic here
    # Output variable assignment
    SET Out_txt_FullyProcessed TO $''''''
END FUNCTION

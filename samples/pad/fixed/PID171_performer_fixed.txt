FUNCTION Main_copy GLOBAL
    # # ── Phase 0: Initialize ──────────────────────────────────────────
    **REGION Initialise Values
    DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> dt_CurrentDateTime
    Text.ConvertDateTimeToText.FromDateTime DateTime: dt_CurrentDateTime StandardFormat: Text.WellKnownDateTimeFormat.SortableDateTime Result=> txt_ProcessStartTime
    SET txt_ExceptionType TO $'''%''%'''
    SET txt_ExceptionMessage TO $'''%''%'''
    SET txt_PreviousExceptionMessage TO $'''%''%'''
    # FIX F7: flg_SendExceptionEmail now means "the System Error path still has
    # to send the exception mail". 'Mark Exception' sets it to False when the
    # consecutive-exception mail was already sent for the same failure.
    SET flg_SendExceptionEmail TO True
    SET flg_Screenshot TO True
    SET txt_LastScreenshotPath TO $'''%''%'''
    SET num_ConsecutiveExcCount TO 0
    SET txt_FailedItemsLog TO $'''%''%'''
    SET num_ItemsException TO 0
    SET num_ItemsCompleted TO 0
    SET obj_Config TO {{ }}
    Variables.ConvertJsonToCustomObject Json: In_txt_Config CustomObject=> obj_Config
    IF IsEmpty(obj_Config) THEN
        SET flg_Screenshot TO False
        SET txt_ExceptionMessage TO $'''No data present in config object input'''
        FlowControl.ThrowCustomError CustomErrorCode: $'''System Exception''' CustomErrorMessage: txt_ExceptionMessage
    END
    # # Load all Config data
    CALL 'Load Config Data'
    # # Close all applications
    CALL 'Close Application' In_txt_AppProcessName: txt_AppProcessName + ',' + txt_ExcelProcessName
    **ENDREGION
    BLOCK 'Get unprocessed items'
ON BLOCK ERROR 'Business Exception' IsUserDefinedErrorCode: True
    SET flg_Screenshot TO False
    CALL 'Get Error'
    SET txt_ExceptionType TO $'''Business Exception'''
    GOTO 'Business Error'
ON BLOCK ERROR
    CALL 'Get Error'
    GOTO 'System Error'
END
        @@'InputSummaryValue:WORKQUEUE': '%WorkQueueId%'
WorkQueues.GetWorkQueueItems WorkQueue: txt_WorkQueueId FilterRows: $'''<attribute name=\"statecode\" />
<filter type=\"and\">
    <condition attribute=\"statecode\" operator=\"eq\" value=\"0\" />
</filter>''' RowsToReturn: 5000 WorkQueueItems=> lst_WorkQueueItems
        IF lst_WorkQueueItems.Count = 0 THEN
            SET txt_ExceptionMessage TO 'No work queue items found in: ' + txt_QueueName
            FlowControl.ThrowCustomError CustomErrorCode: $'''Business Exception''' CustomErrorMessage: txt_ExceptionMessage
        END
    END
    BLOCK 'Launch block'
ON BLOCK ERROR 'System Unavailable Exception' IsUserDefinedErrorCode: True
    # FIX F2: launch/login failures after all retries are System Unavailable -
    # preserve the classification through the System Error path.
    SET flg_Screenshot TO True
    CALL 'Get Error'
    SET txt_ExceptionType TO $'''System Unavailable Exception'''
    GOTO 'System Error'
ON BLOCK ERROR
    SET flg_Screenshot TO True
    CALL 'Get Error'
    GOTO 'System Error'
END
        SET flg_IsAppRunning TO False
        SET flg_IsLoggedIn TO False
        CALL 'Launch Application'
    END
    BLOCK 'Main Work block'
ON BLOCK ERROR 'System Unavailable Exception' IsUserDefinedErrorCode: True
    # FIX F2: rethrown terminal exceptions from 'Process Work Queue Items'
    # keep their classification.
    SET flg_Screenshot TO False
    CALL 'Get Error'
    SET txt_ExceptionType TO $'''System Unavailable Exception'''
    GOTO 'System Error'
ON BLOCK ERROR
    SET flg_Screenshot TO False
    CALL 'Get Error'
    GOTO 'System Error'
END
        CALL 'Process Work Queue Items'
        GOTO 'End'
    END
    LABEL 'System Error'
    CALL 'Close Application' In_txt_AppProcessName: txt_AppProcessName + ',' + txt_ExcelProcessName
    # FIX F7: attach the last captured screenshot (if any) to the exception
    # mail; send only when 'Mark Exception' has not already notified for the
    # same failure (avoids duplicate mails for the same error).
    SET lst_ScreenshotAttachments TO []
    IF txt_LastScreenshotPath <> $'''%''%''' THEN
        IF (File.IfFile.Exists File: txt_LastScreenshotPath) THEN
            Variables.AddItemToList Item: txt_LastScreenshotPath List: lst_ScreenshotAttachments
        END
    END
    IF flg_SendExceptionEmail = True THEN
        CALL 'Send System Exception Mail' In_txt_To_Email: obj_Config['Mail_SystemExceptionTo'] In_txt_Cc_Email: obj_Config['Mail_SystemExceptionCc'] In_txt_Subject: $'''%txt_ProcessName% - System error encountered''' In_txt_ErrorMessage: txt_ExceptionMessage In_lst_Attachments: lst_ScreenshotAttachments
    END
    # FIX F2/F3: return the real exception classification to the parent cloud flow.
    FlowControl.ThrowCustomError CustomErrorCode: txt_ExceptionType CustomErrorMessage: txt_ExceptionMessage
    LABEL 'Business Error'
    GOTO 'End'
    LABEL 'End'
    CALL 'Close Application' In_txt_AppProcessName: txt_AppProcessName + ',' + txt_ExcelProcessName
    SET out_num_CompletedCount TO num_ItemsCompleted
    SET out_num_ExceptionCount TO num_ItemsException
    SET out_txt_ExceptionDetail TO txt_FailedItemsLog
END FUNCTION

FUNCTION 'Get Error' GLOBAL
    # ############### Capture Error Details ###############
    ERROR => LastError
    SET txt_ExceptionType TO $'''System Exception'''
    SET txt_ExceptionMessage TO $'''Error message :%LastError.Message% |
Flow name : %LastError.SubflowName% |
Action name :%LastError.ActionName%'''
    # FIX F7: screenshot capture extracted to 'Capture Error Screenshot' so the
    # exception path can also take ONE final screenshot after retries without
    # duplicating capture logic. txt_LastScreenshotPath is no longer cleared
    # here - it is reset per item in 'Process Work Queue Items'.
    CALL 'Capture Error Screenshot'
END FUNCTION

==================================>

FUNCTION 'Capture Error Screenshot' GLOBAL
    # NEW (FIX F7): captures a single timestamped screenshot into
    # txt_LastScreenshotPath when flg_Screenshot is armed, then disarms the
    # flag so the same failure never produces duplicate screenshots.
    IF flg_Screenshot = True THEN
        DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> dt_CurrentDateTimeVal
        Text.ConvertDateTimeToText.FromCustomDateTime DateTime: dt_CurrentDateTimeVal CustomFormat: $'''MMddyy_hhmmss''' Result=> txt_ScreenShotTime
        SET txt_LastScreenshotPath TO $'''%txt_ScreenshotPath%\\%txt_ScreenShotTime%.png'''
        Workstation.TakeScreenshot.TakeScreenshotAndSaveToFile File: txt_LastScreenshotPath ImageFormat: System.ImageFormat.Png
        ON ERROR
            SET txt_LastScreenshotPath TO $'''%''%'''
        END
        SET flg_Screenshot TO False
    END
END FUNCTION

FUNCTION 'Launch Application' GLOBAL
    IF (System.IfProcess.ProcessIsRunning ProcessName: txt_AppProcessName) THEN
        WAIT (UIAutomation.WaitForWindowContent.WindowToContainElement Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Title Bar']) FOR num_Delay_M
        ON ERROR
            SET flg_IsAppRunning TO False
            SET flg_IsLoggedIn TO False
            GOTO 'Launch application'
        END
        SET flg_IsAppRunning TO True
        EXIT FUNCTION
    END
    LABEL 'Launch application'
    IF flg_IsAppRunning = False THEN
        CALL 'Close Application' In_txt_AppProcessName: txt_AppProcessName
        LOOP num_AppRetryCount FROM 0 TO num_MaxRetryLimit - 1 STEP 1
            BLOCK 'App Start block'
ON BLOCK ERROR all
    SET flg_Screenshot TO True
    CALL 'Get Error'
    CALL 'Close Application' In_txt_AppProcessName: txt_AppProcessName
END
                System.RunApplication.RunApplicationAndWaitToLoad ApplicationPath: obj_Config['App_SampleManagerExePath'] WindowStyle: System.ProcessWindowStyle.Normal Timeout: num_App_WaitTime ProcessId=> txt_AppProcessId
                WAIT (UIAutomation.WaitForWindow.ToOpen Window: appmask['Window \'SampleManager LIMS Login\'']['Title Bar'] FocusWindow: True) FOR num_Delay_L
                SET flg_IsAppRunning TO True
                EXIT LOOP
            END
            WAIT num_Delay_S * num_AppRetryCount
        END
        # FIX F6: the old guard compared the loop counter against
        # num_MaxRetryLimit, which it can never reach (loop ends at limit - 1),
        # so the throw was unreachable and a misleading login error surfaced
        # instead. flg_IsAppRunning alone already proves all retries failed.
        # FIX F2: launch failure after all retries is System Unavailable.
        IF flg_IsAppRunning = False THEN
            FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: $'''Failed to start process \'%txt_AppProcessName%\' after %num_MaxRetryLimit% attempts'''
        END
        IF flg_IsLoggedIn = False THEN
            CALL 'Log In Application' In_txt_AppUsername: txt_AppUsername In_txt_AppPassword: txt_AppPassword Out_flg_Success=> flg_IsLoggedIn
            IF flg_IsLoggedIn = False THEN
                # FIX F2: login failure after all retries is System Unavailable.
                FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: 'Failed to log in to SampleManager after ' + num_MaxRetryLimit + ' attempts - ' + txt_ExceptionMessage
            END
        END
    END
END FUNCTION

FUNCTION 'Set Results Entry' In_txt_ComponentName, In_lst_ComponentNamesList, In_txt_ComponentValue, OUTPUT Out_txt_EntryStatus, OUTPUT Out_txt_EntryError, OUTPUT Out_txt_Unauthorize, OUTPUT Out_txt_OutputValue
    SET txt_ErrorMessage TO $'''%''%'''
    SET Out_txt_OutputValue TO $'''%''%'''
    IF (UIAutomation.IfWindow.IsOpen Window: appmask['Window \'SampleManager LIMS - Explorer\'']['Window \'Result Entry Analysis\''] InstantCheck: False) THEN
        DISABLE UIAutomation.SetWindowState.SetState Window: appmask['Window \'SampleManager LIMS - Explorer\'']['Window \'Result Entry Analysis\''] State: UIAutomation.WindowState.Maximized
        IF (UIAutomation.IfWindowContains.WindowContainsElement Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Window \'Result Entry Analysis\''] InstantCheck: False) THEN
            UIAutomation.Click.Click Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Pane \'Analysis_Chart\''] ClickType: UIAutomation.ClickType.LeftClick MousePositionRelativeToElement: UIAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
            ON ERROR
                CALL 'Get Error'
                SET txt_ErrorMessage TO txt_ErrorMessage + ': ' + GLOBAL.LastError.Message
                GOTO 'Error block'
            END
            LOOP FOREACH txt_ComponentName IN In_lst_ComponentNamesList
                IF In_txt_ComponentValue = $'''%''%''' THEN
                    SET Out_txt_EntryError TO $'''%'Component value is empty'%'''
                    SET Out_txt_Unauthorize TO $'''%'UnAuthorise or Already Authorise'%'''
                    SET Out_txt_EntryStatus TO $'''%'FAIL'%'''
                    GOTO 'Handle Popups'
                END
                Text.ChangeCase Text: In_txt_ComponentName NewCase: Text.CaseOption.LowerCase Result=> In_txt_ComponentName
                IF txt_ComponentName = In_txt_ComponentName THEN
                    UIAutomation.GetElementDetails Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Edit \'Analysis_Value\''] AttributeName: $'''Own Text''' BringToFront: True AttributeValue=> txt_AttributeValue
                    ON ERROR
                        CALL 'Get Error'
                        SET txt_ErrorMessage TO txt_ErrorMessage + 'Unable to read from Component column' + GLOBAL.LastError.Message
                        GOTO 'Error block'
                    END
                    Text.ChangeCase Text: txt_AttributeValue NewCase: Text.CaseOption.LowerCase Result=> txt_AttributeValue
                    IF In_txt_ComponentValue = txt_AttributeValue THEN
                        GOTO Authorize
                    END
                    UIAutomation.PopulateTextField.PopulateTextField TextField: appmask['Window \'SampleManager LIMS - Explorer\'']['Edit \'Analysis_Value\''] Text: In_txt_ComponentValue Mode: UIAutomation.PopulateTextMode.Replace ClickType: UIAutomation.PopulateMouseClickType.SingleClick
                    ON ERROR
                        CALL 'Get Error'
                        SET txt_ErrorMessage TO txt_ErrorMessage + 'Unable to Enter value in Result Entry Analysis Screen' + GLOBAL.LastError.Message
                        GOTO 'Error block'
                    END
                    WAIT (UIAutomation.WaitForWindowContent.WindowToContainText Text: In_txt_ComponentValue Window: appmask['Window \'SampleManager LIMS - Explorer\'']['Edit \'Analysis_Value\''])
                    ON ERROR
                        CALL 'Get Error'
                        SET txt_ErrorMessage TO txt_ErrorMessage + 'Unable to read value in Result Entry Analysis Screen' + GLOBAL.LastError.Message
                        THROW ERROR
                    END
                    # FIX F18: re-read the field AFTER the value was entered so
                    # Out_txt_OutputValue reports the post-entry value, not the
                    # stale pre-entry content (falls back to the previous read
                    # if the re-read fails).
                    UIAutomation.GetElementDetails Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Edit \'Analysis_Value\''] AttributeName: $'''Own Text''' BringToFront: True AttributeValue=> txt_AttributeValue
                    ON ERROR

                    END
                    **REGION Authorise
                    LABEL Authorize
                    UIAutomation.GetElementDetails Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Button \'Authorize\''] AttributeName: $'''Enabled''' BringToFront: True AttributeValue=> flg_Enabled
                    IF flg_Enabled = False THEN
                        GOTO 'Authorise Fail'
                    END
                    UIAutomation.Click.Click Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Button \'Authorize\''] ClickType: UIAutomation.ClickType.LeftClick MousePositionRelativeToElement: UIAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
                    ON ERROR
                        GOTO 'Authorise Fail'
                    END
                    SET Out_txt_EntryStatus TO $'''%'PASS'%'''
                    SET Out_txt_EntryError TO $'''%''%'''
                    SET Out_txt_Unauthorize TO $'''%'Authorise'%'''
                    GOTO 'Handle Popups'
                    **ENDREGION
                    **REGION Authorise Fail
                    LABEL 'Authorise Fail'
                    SET Out_txt_EntryError TO $'''%'Authorise button disabled or Already authorised'%'''
                    SET Out_txt_Unauthorize TO $'''%'UnAuthorise or Already Authorise'%'''
                    SET Out_txt_EntryStatus TO $'''%'FAIL'%'''
                    **ENDREGION
                    **REGION Handle Popups
                    LABEL 'Handle Popups'
                    WAIT (UIAutomation.WaitForWindowContent.WindowToContainElement Element: appmask['Window \'SampleManager\'']) FOR GLOBAL.num_Delay_M
                    ON ERROR
                        GOTO 'Popup #1 end'
                    END
                    IF (UIAutomation.IfWindowContains.WindowContainsElement Element: appmask['Window \'SampleManager\'']['Text \'Modification of existing result\''] InstantCheck: False) THEN
                        UIAutomation.Click.Click Element: appmask['Window \'SampleManager\'']['Button \'Yes\''] ClickType: UIAutomation.ClickType.LeftClick MousePositionRelativeToElement: UIAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
                    END
                    LABEL 'Popup #1 end'
                    WAIT (UIAutomation.WaitForWindowContent.WindowToContainElement Element: appmask['Window \'SampleManager\'']) FOR GLOBAL.num_Delay_M
                    ON ERROR
                        GOTO 'Popup #2 end'
                    END
                    IF (UIAutomation.IfWindowContains.WindowContainsElement Element: appmask['Window \'SampleManager\'']['Text \'Modification of existing result\''] InstantCheck: False) THEN
                        UIAutomation.Click.Click Element: appmask['Window \'SampleManager\'']['Button \'Yes\''] ClickType: UIAutomation.ClickType.LeftClick MousePositionRelativeToElement: UIAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
                        UIAutomation.Click.Click Element: appmask['Window \'SampleManager LIMS - Explorer\'']['Button \'Undo\''] ClickType: UIAutomation.ClickType.LeftClick MousePositionRelativeToElement: UIAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
                        SET Out_txt_EntryStatus TO $'''%'FAIL'%'''
                        SET Out_txt_EntryError TO $'''%'Out Of Range Value'%'''
                        SET Out_txt_Unauthorize TO $'''%'\"UnAuthorise or already Authorise\"'%'''
                    END
                    LABEL 'Popup #2 end'
                    **ENDREGION
                    GOTO 'End'
                ELSE
                    MouseAndKeyboard.SendKeys.FocusAndSendKeysByControl Control: appmask['Window \'SampleManager LIMS - Explorer\'']['Window \'Result Entry Analysis\''] TextToSend: $'''{Down}''' DelayBetweenKeystrokes: 3 SendTextAsHardwareKeys: False
                    ON ERROR
                        CALL 'Get Error'
                        SET txt_ErrorMessage TO txt_ErrorMessage + 'Unable to navigate in Component column' + GLOBAL.LastError.Message
                        GOTO 'Error block'
                    END
                END
            END
            SET Out_txt_EntryError TO $'''%'Parameter Not Matched'%'''
            LABEL 'End'
            SET Out_txt_OutputValue TO txt_AttributeValue
            EXIT FUNCTION
        END
    END
    SET txt_ErrorMessage TO $'''\'Results Entry Analysis\' window not found'''
    LABEL 'Error block'
    FlowControl.ThrowCustomError CustomErrorCode: $'''System Exception''' CustomErrorMessage: txt_ErrorMessage
END FUNCTION

FUNCTION 'Enter Results in App' In_dtb_FinalProduct, txt_SampleId, OUTPUT out_dtb_filteredTable
    **REGION Initialise Variables
    SET txt_ErrorMessage TO $'''%''%'''
    SET num_RowIndex TO 0
    SET num_RunCount TO 0
    SET num_RetryCount TO 0
    Variables.AddColumnToDataTable.AppendColumnToDataTable DataTable: In_dtb_FinalProduct ColumnName: $'''OutputValue'''
    Variables.AddColumnToDataTable.AppendColumnToDataTable DataTable: In_dtb_FinalProduct ColumnName: $'''Unauthorize'''
    Variables.AddColumnToDataTable.AppendColumnToDataTable DataTable: In_dtb_FinalProduct ColumnName: $'''EntryStatus'''
    Variables.AddColumnToDataTable.AppendColumnToDataTable DataTable: In_dtb_FinalProduct ColumnName: $'''EntryError'''
    Variables.CreateNewList List=> lst_MissingIdList
    Variables.CreateNewDatatable InputTable: { } DataTable=> out_dtb_filteredTable
    **ENDREGION
    BLOCK 'Data entry block'
ON BLOCK ERROR
    CALL 'Get Error'
    SET txt_ErrorMessage TO GLOBAL.LastError
    GOTO 'Error block'
END
        LOOP FOREACH dtr_CurrentItem IN In_dtb_FinalProduct
            SET txt_Unauthorize TO $'''%''%'''
            SET txt_OutputValue TO $'''%''%'''
            SET txt_EntryStatus TO $'''%''%'''
            SET txt_EntryError TO $'''%''%'''
            **REGION Data Validation - SampleId, Indentity, Results
            IF txt_SampleId = $'''%''%''' THEN
                SET txt_Unauthorize TO $'''%'NULL'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'NULL'%'''
                SET txt_EntryError TO $'''%'The Sample ID value is empty'%'''
                GOTO 'Summary block'
            END
            IF dtr_CurrentItem['Results'] = $'''%''%''' THEN
                SET txt_Unauthorize TO $'''%'NULL'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'NULL'%'''
                SET txt_EntryError TO $'''%'NULL'%'''
                GOTO 'Summary block'
            END
            IF dtr_CurrentItem['Identity'] = $'''%''%''' THEN
                SET txt_Unauthorize TO $'''%'NULL'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'NULL'%'''
                SET txt_EntryError TO $'''%'The Identity value not found in excel file'%'''
                GOTO 'Summary block'
            END
            Variables.FilterList ListToFilter: lst_MissingIdList FilterParameters: { ^['RULE', 'OPERATOR', 'VALUE'], [$'''''', $'''Equal''', dtr_CurrentItem['Identity']] } IgnoreCase: False FilteredList=> lst_FilteredList
            IF lst_FilteredList.Count > 0 THEN
                # FIX F1: rows for an identity already known to be missing must
                # still receive their summary values AND advance num_RowIndex.
                # The old NEXT LOOP here desynchronised num_RowIndex from the
                # FOREACH row, corrupting every subsequent summary row.
                SET txt_Unauthorize TO $'''%'UnAuthorise'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'FAIL'%'''
                SET txt_EntryError TO GLOBAL.obj_Config['Err_ProductException']
                GOTO 'Summary block'
            END
            **ENDREGION
            SET num_RunCount TO num_RunCount + 1
            IF num_RunCount = 1 THEN
                SET txt_Prev_Identity TO dtr_CurrentItem['Identity']
                CALL 'Open \'Entry By Test\''
                GOTO 'Navigate & Fetch Component List'
            ELSE
                IF dtr_CurrentItem['Identity'] = txt_Prev_Identity THEN
                    SET txt_Prev_Identity TO dtr_CurrentItem['Identity']
                    IF txt_EntryError = GLOBAL.obj_Config['Err_ProductException'] THEN
                        SET num_RowIndex TO num_RowIndex + 1
                        NEXT LOOP
                    ELSE
                        GOTO 'Data Entry to SampleManager'
                    END
                ELSE
                    DISABLE GOTO 'Summary block'
                    CALL 'Close Results Entry Analysis'
                    CALL 'Open \'Entry By Test\''
                END
            END
            **REGION Navigate to Results Analysis & Fetch component Name
            LABEL 'Navigate & Fetch Component List'
            SET flg_Success TO False
            BLOCK 'Fetch Sample ID block'
ON BLOCK ERROR
    CALL 'Get Error'
    SET txt_ErrorMessage TO txt_ErrorMessage + ': ' + GLOBAL.LastError.Message
END
                CALL 'Get Results by Analysis and SampleId' in_txt_Analysis: dtr_CurrentItem['Identity'] in_txt_SampleId: txt_SampleId
                SET flg_Success TO True
            END
            IF flg_Success = False THEN
                Variables.AddItemToList Item: dtr_CurrentItem['Identity'] List: lst_MissingIdList
                SET txt_Unauthorize TO $'''%'UnAuthorise'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'FAIL'%'''
                SET txt_EntryError TO GLOBAL.obj_Config['Err_ProductException']
                GOTO 'Summary block'
            END
            LOOP num_RetryCount FROM 1 TO GLOBAL.num_MaxRetryLimit STEP 1
                BLOCK 'Component names block'
ON BLOCK ERROR
    CALL 'Get Error'
    SET txt_ErrorMessage TO txt_ErrorMessage + ': ' + GLOBAL.LastError.Message
END
                    CALL 'Get Components List' out_lst_ComponentNames=> lst_ComponentNames
                    IF lst_ComponentNames.Count > 0 THEN
                        LOOP num_LoopIndex FROM 0 TO lst_ComponentNames.Count - 1 STEP 1
                            Text.ChangeCase Text: lst_ComponentNames[num_LoopIndex] NewCase: Text.CaseOption.LowerCase Result=> txt_CurrentComponent
                            SET lst_ComponentNames[num_LoopIndex] TO txt_CurrentComponent
                        END
                        EXIT LOOP
                    END
                END
            END
            IF lst_ComponentNames.Count = 0 THEN
                GOTO 'Error block'
            END
            **ENDREGION
            SET num_RetryCount TO 0
            **REGION Data Entry to SampleManager
            LABEL 'Data Entry to SampleManager'
            IF dtr_CurrentItem['Component'] = $'''%''%''' THEN
                SET txt_Unauthorize TO $'''%'NULL'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'FAIL'%'''
                SET txt_EntryError TO $'''%'The Component value not found in excel file'%'''
                GOTO 'Summary block'
            END
            IF dtr_CurrentItem['Results'] = $'''%''%''' THEN
                SET txt_Unauthorize TO $'''%'UnAuthorise'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'FAIL'%'''
                SET txt_EntryError TO $'''%'The Identity value not found in excel file'%'''
                GOTO 'Summary block'
            END
            SET flg_Success TO False
            LOOP num_RetryCount FROM 1 TO GLOBAL.num_MaxRetryLimit STEP 1
                BLOCK 'Component data entry block'
ON BLOCK ERROR
    CALL 'Get Error'
    SET txt_ErrorMessage TO txt_ErrorMessage + ': ' + GLOBAL.LastError.Message
END
                    # FIX F20: if a previous attempt lost the 'Result Entry
                    # Analysis' window, re-navigate before retrying instead of
                    # blindly re-calling into a missing window (BP recovered
                    # via Sample Manager - Explorer -> Result Entry).
                    IF (UIAutomation.IfWindow.IsNotOpen Window: appmask['Window \'SampleManager LIMS - Explorer\'']['Window \'Result Entry Analysis\''] InstantCheck: False) THEN
                        CALL 'Get Results by Analysis and SampleId' in_txt_Analysis: dtr_CurrentItem['Identity'] in_txt_SampleId: txt_SampleId
                    END
                    CALL 'Set Results Entry' In_txt_ComponentName: dtr_CurrentItem['Component'] In_lst_ComponentNamesList: lst_ComponentNames In_txt_ComponentValue: dtr_CurrentItem['Results'] Out_txt_EntryStatus=> txt_EntryStatus Out_txt_EntryError=> txt_EntryError Out_txt_Unauthorize=> txt_Unauthorize Out_txt_OutputValue=> txt_OutputValue
                    SET flg_Success TO True
                    EXIT LOOP
                END
            END
            IF flg_Success = False THEN
                GOTO 'Error block'
            END
            IF txt_EntryError = $'''Parameter Not Matched''' THEN
                SET txt_EntryError TO 'Component (' + dtr_CurrentItem['Component'] + ') not found in Sample Manager'
                SET txt_Unauthorize TO $'''%'UnAuthorise'%'''
                SET txt_OutputValue TO $'''%'NULL'%'''
                SET txt_EntryStatus TO $'''%'FAIL'%'''
                GOTO 'Summary block'
            END
            IF txt_OutputValue = $'''%''%''' THEN
                SET txt_OutputValue TO $'''%'NULL'%'''
            END
            **ENDREGION
            LABEL 'Summary block'

FUNCTION 'Mark Complete' In_obj_WorkQueueItem
    SET obj_WorkQueueItem TO In_obj_WorkQueueItem
    DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> dt_CurrentDateTime
    Text.ConvertDateTimeToText.FromDateTime DateTime: dt_CurrentDateTime StandardFormat: Text.WellKnownDateTimeFormat.SortableDateTime Result=> txt_FormattedDateTime
    # FIX EB-2 (corrected): ON BLOCK ERROR handlers cannot carry conditional
    # logic - PAD runs the handler body as a fixed action sequence, not a
    # branch (see 'Launch Application's flg_IsAppRunning pattern: the
    # retries-exhausted decision is always made AFTER the loop, never inside
    # the handler). The handler below is a pure swallow (do nothing, let the
    # loop retry) - it deliberately does NOT call 'Get Error', since that
    # would overwrite GLOBAL.txt_ExceptionType / txt_ExceptionMessage with the
    # QUEUE UPDATE's own error and clobber the item's exception context.
    SET flg_QueueUpdateSuccess TO False
    LOOP num_QueueUpdateRetryCount FROM 1 TO GLOBAL.num_MaxRetryLimit STEP 1
        BLOCK 'Queue update block - Complete'
ON BLOCK ERROR

END
            WorkQueues.UpdateProcessingNotes.WithProcessingNotes WorkQueueItem: obj_WorkQueueItem ProcessingNotes: 'Processing completed at ' + txt_FormattedDateTime
            WorkQueues.UpdateWorkQueueItem.UpdateWithProcessingNotes WorkQueueItem: obj_WorkQueueItem Status: WorkQueues.WorkQueueItemStatus.Processed
            SET flg_QueueUpdateSuccess TO True
            EXIT LOOP
        END
        WAIT GLOBAL.num_Delay_S * num_QueueUpdateRetryCount
    END
    IF flg_QueueUpdateSuccess = False THEN
        # NOTE (open question, see fix notes): unlike EB-1, a Mark-Complete
        # failure after retries bubbles straight to Main_copy's global
        # handler - the item is NOT explicitly routed through 'Mark
        # Exception'. Left as-is pending a decision on whether to close this
        # symmetric gap.
        FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: 'Unable to mark work queue item complete after ' + GLOBAL.num_MaxRetryLimit + ' attempts.'
    END
    # FIX F15: success/exception counters are maintained with plain SET in
    # 'Process Work Queue Items' (GLOBAL scope). Variables.IncreaseVariable on
    # a GLOBAL.-prefixed variable from a non-global subflow fails because the
    # action resolves its target in the local scope - which is why the old
    # DISABLEd increment errored here.
END FUNCTION

FUNCTION 'Mark Exception' In_obj_WorkQueueItem
    SET obj_WorkQueueItem TO In_obj_WorkQueueItem
    DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> dt_CurrentDateTime
    Text.ConvertDateTimeToText.FromDateTime DateTime: dt_CurrentDateTime StandardFormat: Text.WellKnownDateTimeFormat.SortableDateTime Result=> txt_FormattedDateTime
    # FIX F7: attach the single final screenshot (captured once by
    # 'Capture Error Screenshot' after all retries) - never duplicates.
    SET lst_ScreenshotAttachments TO []
    IF GLOBAL.txt_LastScreenshotPath <> $'''%''%''' THEN
        IF (File.IfFile.Exists File: GLOBAL.txt_LastScreenshotPath) THEN
            Variables.AddItemToList Item: GLOBAL.txt_LastScreenshotPath List: lst_ScreenshotAttachments
        END
    END
    # FIX EB-2 (corrected): same pattern as 'Mark Complete' above - empty
    # (swallow) handlers, a success flag, and the retries-exhausted decision
    # made AFTER each loop. None of these handlers call 'Get Error', so
    # GLOBAL.txt_ExceptionType / txt_ExceptionMessage (the ITEM's original
    # exception, read below for ProcessingResult and mail bodies) are never
    # touched by a queue-update retry.
    IF GLOBAL.txt_ExceptionType.ToLower = $'''business exception''' THEN
        SET flg_QueueUpdateSuccess TO False
        LOOP num_QueueUpdateRetryCount FROM 1 TO GLOBAL.num_MaxRetryLimit STEP 1
            BLOCK 'Queue update block - BE'
ON BLOCK ERROR

END
                WorkQueues.UpdateWorkQueueItem.UpdateWithProcessingNotes WorkQueueItem: obj_WorkQueueItem Status: WorkQueues.WorkQueueItemStatus.BusinessException ProcessingResult: GLOBAL.txt_ExceptionMessage
                SET flg_QueueUpdateSuccess TO True
                EXIT LOOP
            END
            WAIT GLOBAL.num_Delay_S * num_QueueUpdateRetryCount
        END
        IF flg_QueueUpdateSuccess = False THEN
            FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: 'Unable to update work queue item status after ' + GLOBAL.num_MaxRetryLimit + ' attempts.'
        END
        CALL 'Send Business Exception Mail' In_txt_To_Email: GLOBAL.obj_Config['Mail_BusinessExceptionTo'] In_txt_Cc_Email: GLOBAL.obj_Config['Mail_BusinessExceptionCc'] In_txt_Subject: $'''%GLOBAL.txt_ProcessName% - Business Error''' In_txt_ErrorMessage: GLOBAL.txt_ExceptionMessage In_lst_Attachments: []
    ELSE IF GLOBAL.txt_ExceptionType.ToLower = $'''system unavailable exception''' THEN
        # FIX F2: now reachable - the typed handler clauses in
        # 'Process Work Queue Items' preserve the custom error code. The
        # notification (with screenshot) is sent once by Main_copy's
        # 'System Error' path after this item is marked, so
        # flg_SendExceptionEmail stays True here.
        SET flg_QueueUpdateSuccess TO False
        LOOP num_QueueUpdateRetryCount FROM 1 TO GLOBAL.num_MaxRetryLimit STEP 1
            BLOCK 'Queue update block - SUE'
ON BLOCK ERROR

END
                WorkQueues.UpdateWorkQueueItem.UpdateWithProcessingNotes WorkQueueItem: obj_WorkQueueItem Status: WorkQueues.WorkQueueItemStatus.ITException ProcessingResult: GLOBAL.txt_ExceptionMessage
                SET flg_QueueUpdateSuccess TO True
                EXIT LOOP
            END
            WAIT GLOBAL.num_Delay_S * num_QueueUpdateRetryCount
        END
        IF flg_QueueUpdateSuccess = False THEN
            FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: 'Unable to update work queue item status after ' + GLOBAL.num_MaxRetryLimit + ' attempts.'
        END
    ELSE
        # FIX F3: a System Exception only reaches here after
        # num_ConsecutiveExcLimit in-place attempts on the same item, so this
        # is the consecutive-failure terminal path (BP: TERMINATE). The
        # consecutive mail replaces the generic system mail - disarm the
        # System Error path's mail to avoid a duplicate notification.
        SET flg_QueueUpdateSuccess TO False
        LOOP num_QueueUpdateRetryCount FROM 1 TO GLOBAL.num_MaxRetryLimit STEP 1
            BLOCK 'Queue update block - SE'
ON BLOCK ERROR

END
                WorkQueues.UpdateWorkQueueItem.UpdateWithProcessingNotes WorkQueueItem: obj_WorkQueueItem Status: WorkQueues.WorkQueueItemStatus.GenericException ProcessingResult: GLOBAL.num_ConsecutiveExcLimit + ' consecutive incidents of ' + GLOBAL.txt_ExceptionType + ': ' + GLOBAL.txt_ExceptionMessage
                SET flg_QueueUpdateSuccess TO True
                EXIT LOOP
            END
            WAIT GLOBAL.num_Delay_S * num_QueueUpdateRetryCount
        END
        IF flg_QueueUpdateSuccess = False THEN
            FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: 'Unable to update work queue item status after ' + GLOBAL.num_MaxRetryLimit + ' attempts.'
        END
        CALL 'Send Consecutive Exception Mail' In_txt_To_Email: GLOBAL.obj_Config['Mail_SystemExceptionTo'] In_txt_Cc_Email: GLOBAL.obj_Config['Mail_SystemExceptionCc'] In_txt_Subject: $'''%GLOBAL.txt_ProcessName% - Consecutive error encountered''' In_txt_ErrorMessage: GLOBAL.txt_ExceptionMessage In_lst_Attachments: lst_ScreenshotAttachments
        SET GLOBAL.flg_SendExceptionEmail TO False
    END
END FUNCTION

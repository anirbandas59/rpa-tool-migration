        GOTO 'End'
    END
    # # ── Phase 2: Per-Email Processing (isolated, hash-keyed, scoped folders) ──
    BLOCK 'Add to Queue Block'
ON BLOCK ERROR all
    SET flg_Screenshot TO False
    CALL 'Get Error'
    GOTO 'Error Block'
END
        LOOP FOREACH obj_CurrentMail IN lst_MailItems
            # # --- 2a: Move mails with attachment to process folder, else move to exception folder ---

                @@folderPath: '%txt_MailExceptionFolder%'
@@connectionDisplayName: 'Office 365 Outlook'

    GOTO 'Next loop - Mail Items'

                @@'InputSummaryValue:WORKQUEUE': '%obj_Config[\'Ctrl_WorkQueueId\']%'
WorkQueues.GetWorkQueueItems WorkQueue: obj_Config['Ctrl_WorkQueueId'] FilterRows: $'''<attribute name=\"name\" />

            # FIX F9: reject payloads that exceed the work-queue-item Value
            # capacity BEFORE enqueue, treat like an invalid mail (notify +

                @@folderPath: '%txt_MailExceptionFolder%'

External.InvokeCloudConnector Connection: '480424f2-a42e-4e13-9e4f-5c212c2dfa3c' ConnectorId: '/providers/Microsoft.PowerApps/apis/shared_office365' OperationId: 'MoveV2' @messageId: obj_CurrentMail.id @folderPath: txt_MailExceptionFolder @mailboxAddress: obj_Config['Mail_MailboxName'] @MoveV2Response=> obj_MailResponse

                @@'InputSummaryValue:WORKQUEUE': '%obj_Config[\'Ctrl_WorkQueueId\']%'

FUNCTION 'Get Error' GLOBAL
    # ############### Capture Error Details ###############
    ERROR => obj_LastError
    SET txt_ExceptionType TO $'''System Exception'''
    SET txt_ExceptionMessage TO $'''Error message :%obj_LastError.Message% |
Flow name : %obj_LastError.SubflowName% |
Action name :%obj_LastError.ActionName%'''
    # ############### Capture Screenshot ###############
    IF flg_Screenshot = True THEN
        DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> dt_CurrentDateTimeVal
        Text.ConvertDateTimeToText.FromCustomDateTime DateTime: dt_CurrentDateTimeVal CustomFormat: $'''MMddyy_hhmmss''' Result=> txt_ScreenShotTime
        Workstation.TakeScreenshot.TakeScreenshotAndSaveToFile File: $'''%txt_ScreenshotPath%\\%txt_ScreenShotTime%.png''' ImageFormat: System.ImageFormat.Png
        ON ERROR

        END
        SET flg_Screenshot TO False
    END
END FUNCTION

FUNCTION 'Send Business Exception Mail' In_txt_To_Email, In_txt_Cc_Email, In_txt_Subject, In_txt_ErrorMessage, In_lst_Attachments
    SET txt_ErrorMessage TO $'''%''%'''
    DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> dt_CurrentDateTime
    Text.ConvertDateTimeToText.FromDateTime DateTime: dt_CurrentDateTime StandardFormat: Text.WellKnownDateTimeFormat.LongDate Result=> txt_FormattedDateTime
    SET txt_MailBody TO GLOBAL.obj_Config['Mail_EmailBodyBusinessException']
    Text.Replace.ReplaceText Text: txt_MailBody TextToFind: $'''{{RunDate}}''' IgnoreCase: False ReplaceWith: txt_FormattedDateTime ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailBody
    Text.Replace.ReplaceText Text: txt_MailBody TextToFind: $'''{{ErrorMessage}}''' IgnoreCase: False ReplaceWith: In_txt_ErrorMessage ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailBody
    Text.Replace.ReplaceText Text: txt_MailBody TextToFind: $'''{{QueueName}}''' IgnoreCase: False ReplaceWith: GLOBAL.obj_Config['Ctrl_QueueName'] ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailBody
    Text.Replace.ReplaceText Text: txt_MailBody TextToFind: $'''{{ResourceName}}''' IgnoreCase: False ReplaceWith: GLOBAL.txt_ResourceName ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailBody
    Text.Replace.ReplaceText Text: txt_MailBody TextToFind: $'''{{Username}}''' IgnoreCase: False ReplaceWith: GLOBAL.txt_UserName ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailBody
    Text.Replace.ReplaceText Text: txt_MailBody TextToFind: $'''{{ProcessName}}''' IgnoreCase: False ReplaceWith: GLOBAL.txt_ProcessName ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailBody
    IF (In_txt_To_Email + In_txt_Cc_Email) = $'''%''%''' THEN
        FlowControl.ThrowCustomError CustomErrorCode: $'''Business Exception''' CustomErrorMessage: $'''Recipients cannot be empty or blank'''
    END
    IF In_txt_Subject = $'''%''%''' THEN
        FlowControl.ThrowCustomError CustomErrorCode: $'''Business Exception''' CustomErrorMessage: $'''Subject cannot be empty or blank'''
    END
    Variables.CreateNewList List=> lst_AttachmentsList
    SET lst_AttachmentsList TO []
    IF In_lst_Attachments.Count > 0 THEN
        LOOP FOREACH CurrentItem IN In_lst_Attachments
            File.GetPathPart File: CurrentItem RootPath=> txt_RootPath Directory=> txt_Directory FileName=> txt_FileName FileNameWithoutExtension=> FileNameNoExtension Extension=> txt_FileExtension
            File.ConvertToBase64 File: CurrentItem Base64Text=> txt_Base64Text
            SET obj_AttachmentObject TO { 'Name': txt_FileName, 'ContentBytes': txt_Base64Text }
            Variables.AddItemToList Item: obj_AttachmentObject List: lst_AttachmentsList
        END
    END
    @@'emailMessage/Sensitivity': ''
@@connectionDisplayName: 'Office 365 Outlook'
External.InvokeCloudConnector Connection: '480424f2-a42e-4e13-9e4f-5c212c2dfa3c' ConnectorId: '/providers/Microsoft.PowerApps/apis/shared_office365' OperationId: 'SendEmailV2' @'emailMessage/To': In_txt_To_Email @'emailMessage/Subject': In_txt_Subject @'emailMessage/Body': txt_MailBody @'emailMessage/From': GLOBAL.obj_Config['Mail_MailboxName'] @'emailMessage/Cc': In_txt_Cc_Email @'emailMessage/Attachments': lst_AttachmentsList @'emailMessage/Importance': $'''Normal'''
    ON ERROR
        CALL 'Get Error'
        SET txt_ErrorMessage TO 'Could not send email - ' + GLOBAL.obj_LastError.Message
        GOTO 'Error File Process'
    END
    EXIT FUNCTION
    LABEL 'Error File Process'
    FlowControl.ThrowCustomError CustomErrorCode: $'''Business Exception''' CustomErrorMessage: txt_ErrorMessage
END FUNCTION

FUNCTION 'Load Config Data' GLOBAL
    BLOCK 'Assign values from config'
ON BLOCK ERROR all
    SET flg_Screenshot TO False
    SET flg_ConfigError TO True
    CALL 'Get Error'
    THROW ERROR
END
        # ####################### Get Computer name and User name #######################
        System.GetEnvironmentVariable.GetEnvironmentVariable Name: $'''USERNAME''' Value=> txt_UserName
        SET txt_ProcessName TO obj_Config['Ctrl_ProcessName']
        System.GetEnvironmentVariable.GetEnvironmentVariable Name: $'''COMPUTERNAME''' Value=> txt_ResourceName
        Text.Trim Text: txt_ProcessName TrimOption: Text.TrimOption.Both TrimmedText=> txt_ProcessName
        Text.Trim Text: txt_ResourceName TrimOption: Text.TrimOption.Both TrimmedText=> txt_ResourceName
        # ####################### Get Limits and Wait times #######################
        Text.ToNumber Text: obj_Config['App_WaitTime'] Number=> num_App_WaitTime
        Text.ToNumber Text: obj_Config['App_MailFetchLimit'] Number=> num_MailFetchLimit
        SET num_MailRunCount TO 0
        Text.ToNumber Text: obj_Config['Ctrl_ConsecutiveExceptionLimit'] Number=> num_MailRetryLimit
        SET lst_MailItems TO []
        # ####################### Set mail folder paths #######################
        SET txt_MailProcessFolder TO obj_Config['Mail_SubFolder']
        SET txt_MailCompletedFolder TO obj_Config['Mail_OutlookFolder'] + '/' + obj_Config['Mail_EmailCompleted']
        SET txt_MailExceptionFolder TO obj_Config['Mail_OutlookFolder'] + '/' + obj_Config['Mail_EmailException']
        Text.Replace.ReplaceText Text: txt_MailCompletedFolder TextToFind: $'''\\''' IgnoreCase: False ReplaceWith: $'''/''' ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailCompletedFolder
        Text.Replace.ReplaceText Text: txt_MailExceptionFolder TextToFind: $'''\\''' IgnoreCase: False ReplaceWith: $'''/''' ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_MailExceptionFolder
        # ####################### Set local folder paths #######################
        Text.Replace.ReplaceText Text: obj_Config['Path_RootFolder'] TextToFind: $'''<<username>>''' IgnoreCase: False ReplaceWith: txt_UserName ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_RootPath
        IF (Folder.IfFolderExists.DoesNotExist Path: txt_RootPath) THEN
            SET txt_ExceptionMessage TO 'Folder not found: ' + txt_RootPath
            FlowControl.ThrowCustomError CustomErrorCode: $'''System Exception''' CustomErrorMessage: txt_ExceptionMessage
        END
        Text.Replace.ReplaceText Text: obj_Config['Path_ProjectFolder'] TextToFind: $'''<<processname>>''' IgnoreCase: False ReplaceWith: txt_ProcessName ActivateEscapeSequences: False ComparisonType: Text.TextComparisonType.CultureSensitive Result=> txt_ProjectFolder
        IF (Folder.IfFolderExists.DoesNotExist Path: txt_RootPath + txt_ProjectFolder) THEN
            Text.CropText.CropTextAfterFlag Text: txt_ProjectFolder FromFlag: $'''\\''' IgnoreCase: False CroppedText=> txt_NewFolder
            Folder.Create FolderPath: txt_RootPath FolderName: txt_NewFolder Folder=> txt_NewFolder
        END
        SET txt_DownloadPath TO $'''%txt_RootPath%%txt_ProjectFolder%%obj_Config['Path_EmailDownload']%'''
        IF (Folder.IfFolderExists.DoesNotExist Path: txt_DownloadPath) THEN
            Text.CropText.CropTextAfterFlag Text: txt_ProjectFolder FromFlag: $'''\\''' IgnoreCase: False CroppedText=> txt_NewFolder
            Folder.Create FolderPath: txt_RootPath FolderName: $'''%txt_NewFolder%%obj_Config['Path_EmailDownload']%''' Folder=> txt_NewFolder
        END
        SET txt_SummaryReportPath TO $'''%txt_RootPath%%txt_ProjectFolder%%obj_Config['Path_SummaryReport']%'''
        IF (Folder.IfFolderExists.DoesNotExist Path: txt_SummaryReportPath) THEN
            Text.CropText.CropTextAfterFlag Text: txt_ProjectFolder FromFlag: $'''\\''' IgnoreCase: False CroppedText=> txt_NewFolder
            Folder.Create FolderPath: txt_RootPath FolderName: $'''%txt_NewFolder%%obj_Config['Path_SummaryReport']%''' Folder=> txt_NewFolder
        END
        SET txt_ScreenshotPath TO $'''%txt_RootPath%%txt_ProjectFolder%%obj_Config['Path_ScreenshotFolder']%'''
        IF (Folder.IfFolderExists.DoesNotExist Path: txt_ScreenshotPath) THEN
            Text.CropText.CropTextAfterFlag Text: txt_ProjectFolder FromFlag: $'''\\''' IgnoreCase: False CroppedText=> txt_NewFolder
            Folder.Create FolderPath: txt_RootPath FolderName: $'''%txt_NewFolder%%obj_Config['Path_ScreenshotFolder']%''' Folder=> txt_NewFolder
        END
    END
END FUNCTION

FUNCTION 'Fetch Emails from Mailbox' GLOBAL
    SET flg_Success TO False
    LOOP num_MailRunCount FROM 1 TO num_MailRetryLimit STEP 1
        BLOCK 'Get Email block'
ON BLOCK ERROR
    # FIX F4/F2: retry Outlook API failures up to num_MailRetryLimit attempts
    # (no logging per design decision), then raise System Unavailable
    # Exception. The old handler rethrew on the FIRST failure, defeating the
    # retry loop.
    SET flg_Screenshot TO False
    CALL 'Get Error'
    IF num_MailRunCount >= num_MailRetryLimit THEN
        FlowControl.ThrowCustomError CustomErrorCode: $'''System Unavailable Exception''' CustomErrorMessage: 'Outlook is not reachable after ' + num_MailRetryLimit + ' attempts - ' + obj_LastError.Message
    END
END
            @@folderPath: '%obj_Config[\'Mail_OutlookFolder\']%'
@@connectionDisplayName: 'Office 365 Outlook'
External.InvokeCloudConnector Connection: '480424f2-a42e-4e13-9e4f-5c212c2dfa3c' ConnectorId: '/providers/Microsoft.PowerApps/apis/shared_office365' OperationId: 'GetEmailsV3' @folderPath: obj_Config['Mail_OutlookFolder'] @fetchOnlyWithAttachment: False @subjectFilter: obj_Config['Mail_EmailSubject'] @fetchOnlyUnread: True @mailboxAddress: obj_Config['Mail_MailboxName'] @includeAttachments: True @top: num_MailFetchLimit @GetEmailsV3Response=> obj_ReceivedEmails
            IF obj_ReceivedEmails.value.Count > 0 THEN
                SET lst_MailItems TO obj_ReceivedEmails.value
                LOOP num_ItemIndex FROM lst_MailItems.Count - 1 TO 0 STEP -1
                    SET txt_FromAddress TO lst_MailItems[num_ItemIndex]['from']
                    IF txt_FromAddress <> obj_Config['Mail_SenderMailId'] THEN
                        Variables.RemoveItemFromList.RemoveItemFromListByIndex ItemIndex: num_ItemIndex List: lst_MailItems
                    END
                END
                SET flg_Success TO True
                EXIT LOOP
            END
        END
        # FIX F5: incremental back-off between attempts. The old code multiplied
        # by num_ConnectionRetryCount which was initialised to 0 and never
        # incremented, so every retry waited 0 seconds.
        WAIT num_MailRunCount * num_App_WaitTime
    END
END FUNCTION
